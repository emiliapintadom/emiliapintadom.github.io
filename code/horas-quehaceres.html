R · IMCO Women in the Economy monitor

Weekly hours of housework, young employed women in Mexico City

Emilia Pintado Mariscal2025R · survey · srvyr · dplyr · purrr
  1. Reads the four 2025 quarters of INEGI's employment survey (ENOE) and joins the sociodemographic table with the employment questionnaire, one-to-one.
  2. Handles changes between quarters: renamed geographic variables and a question numbered differently in the first quarter.
  3. Builds one survey design per quarter (clusters, strata, weights) and averages the four estimates, combining their standard errors.
  4. Estimates for Mexico City, its 16 boroughs and the country, and flags borough figures whose coefficient of variation makes them unpublishable.
# =============================================================================
# == HORAS A LA SEMANA EN QUEHACERES DOMÉSTICOS — mujeres ocupadas de 15 a 29 años - ENOE ==
# =============================================================================

library(survey)
library(dplyr)
library(srvyr)
library(purrr)

# PARÁMETROS

enoe_dir   <- "datos/ENOE"
out_dir    <- "resultados"
trimestres <- c("T125", "T225", "T325", "T425")

# Llaves para unir SDEMT (tabla sociodemográfica) con COE2T (ocupación y empleo)
llaves <- c("tipo", "mes_cal", "cd_a", "ent", "con", "v_sel", "n_hog", "h_mud", "n_ren")

# LECTURA DE MICRODATOS

# Función para cargar y preparar cada base (SDEMT + COE2T del mismo trimestre)
cargar_trimestre <- function(tri) {
  path      <- file.path(enoe_dir, tri, paste0("ENOE_SDEM", tri, ".csv"))
  path_coe2 <- file.path(enoe_dir, tri, paste0("ENOE_COE2", tri, ".csv"))

  df   <- read.csv(path)
  coe2 <- read.csv(path_coe2)
  names(coe2) <- tolower(names(coe2))

  # A partir del 3T2025 se renombraron las variables geográficas cve_
  if ("cve_ent" %in% names(df)) {
    df <- df %>%
      rename(ent = cve_ent, mun = cve_mun, loc = cve_loc, ageb = cve_ageb)
  }
  if ("cve_ent" %in% names(coe2)) {
    coe2 <- coe2 %>%
      rename(ent = cve_ent)
  }

  # La pregunta de actividades sin pago se llama p11 en el 1er trimestre y p9 en los demás, aquí le pongo p9 a todas.
  # Opción 7 = quehaceres de su hogar: p9_7 (lo reportó), p9_h7 (horas), p9_m7 (minutos)
  if ("p11_7" %in% names(coe2)) {
    coe2 <- coe2 %>%
      select(all_of(llaves), p11_7, p11_h7, p11_m7) %>%
      rename(p9_7 = p11_7, p9_h7 = p11_h7, p9_m7 = p11_m7)
  } else {
    coe2 <- coe2 %>%
      select(all_of(llaves), p9_7, p9_h7, p9_m7)
  }

  # Verificar que cada persona aparezca en las dos tablas
  coe2 <- coe2 %>%
    mutate(en_coe2 = TRUE)

  df <- df %>%
    left_join(coe2, by = llaves, relationship = "one-to-one") %>%
    mutate(
      eda       = as.numeric(eda),
      c_res     = as.numeric(c_res),
      TRIMESTRE = tri
    )

  return(df)
}

# Leer y concatenar todas las bases
sdem_todos <- map_dfr(trimestres, cargar_trimestre)

# Horas a la semana en quehaceres
# 98 y 99 en p9_h7 son códigos de no sabe / no especificado, no horas: quedan como NA
sdem_todos <- sdem_todos %>%
  mutate(
    quehaceres     = !is.na(p9_7),
    hrs_quehaceres = ifelse(p9_h7 %in% c(98, 99), NA, p9_h7 + p9_m7 / 60)
  )

# DISEÑO MUESTRAL

options(survey.lonely.psu = "adjust")

# Filtrar población
sdem_filtered <- sdem_todos %>%
  filter(
    r_def  == 0,
    clase2 == 1,            # ocupadas
    eda >= 15 & eda <= 29,
    c_res %in% c(1, 3),
    ent == 9
  )

# Población nacional (mismo filtro de ocupación/edad, sin restringir a ent == 9)
sdem_filtered_nacional <- sdem_todos %>%
  filter(
    r_def  == 0,
    clase2 == 1,
    eda >= 15 & eda <= 29,
    c_res %in% c(1, 3)
  )

# Un diseño muestral por trimestre, con el fac_tri original (sin dividir)
disenar <- function(df) {
  df %>%
    as_survey_design(
      ids     = upm,
      strata  = est_d_tri,
      nest    = TRUE,
      weights = fac_tri
    )
}

sdemsvy_tri          <- sdem_filtered          %>% split(.$TRIMESTRE) %>% map(disenar)
sdemsvy_tri_nacional <- sdem_filtered_nacional %>% split(.$TRIMESTRE) %>% map(disenar)

# Promedio anual: cada trimestre se estima por separado, se divide entre 4 y
# después se suman. El error estándar se combina suponiendo trimestres
# independientes: sqrt(sum(se^2)) / 4
anual <- function(disenos, nombre, estimar, grupos = character()) {
  res <- map_dfr(disenos, estimar, .id = "TRIMESTRE") %>%
    group_by(across(all_of(grupos))) %>%
    summarise(
      est    = sum(est / 4),
      est_se = sqrt(sum((est_se / 4)^2)),
      n_tri  = n(),
      .groups = "drop"
    )
  if (any(res$n_tri < 4)) warning(nombre, ": algún grupo no aparece en los 4 trimestres")
  res %>%
    select(-n_tri) %>%
    rename(!!nombre := est, !!paste0(nombre, "_se") := est_se)
}

# TABLAS DINÁMICAS

# Horas promedio en quehaceres de las mujeres ocupadas que los hacen, CDMX
Hrs_quehaceres <- anual(sdemsvy_tri, "hrs_quehaceres", function(d) d %>%
  filter(sex == 2, quehaceres) %>%
  summarise(est = survey_mean(hrs_quehaceres, na.rm = TRUE)))

# NACIONAL

Hrs_quehaceres_nacional <- anual(sdemsvy_tri_nacional, "hrs_quehaceres", function(d) d %>%
  filter(sex == 2, quehaceres) %>%
  summarise(est = survey_mean(hrs_quehaceres, na.rm = TRUE)))

# ALCALDÍAS

  `4`  = "Cuajimalpa de Morelos",
  `5`  = "Gustavo A. Madero",
  `6`  = "Iztacalco",
  `7`  = "Iztapalapa",
  `8`  = "La Magdalena Contreras",
  `9`  = "Milpa Alta",
  `10` = "Álvaro Obregón",
  `11` = "Tláhuac",
  `12` = "Tlalpan",
  `13` = "Xochimilco",
  `14` = "Benito Juárez",
  `15` = "Cuauhtémoc",
  `16` = "Miguel Hidalgo",
  `17` = "Venustiano Carranza"

Hrs_quehaceres_x_alcaldia <- anual(sdemsvy_tri, "hrs_quehaceres", grupos = "mun", function(d) d %>%
  filter(sex == 2, quehaceres) %>%
  group_by(mun) %>%
  summarise(est = survey_mean(hrs_quehaceres, na.rm = TRUE))) %>%
  mutate(alcaldia = alcaldias_cdmx[as.character(mun)]) %>%
  relocate(alcaldia, .after = mun)

# REVISIÓN EN CONSOLA (no se exporta)

# Registros del universo sin par en COE2T, por trimestre (debe salir 0)
print(sdem_filtered %>%
        group_by(TRIMESTRE) %>%
        summarise(sin_coe2 = sum(is.na(en_coe2))))

# Horas promedio, error estándar, CV y observaciones sin ponderar, CDMX
# n_num: ocupadas que hacen quehaceres con horas válidas; n_den: todas las ocupadas
# n_codigo: reportaron quehaceres pero con 98/99 en horas (quedan fuera del promedio)
Hrs_quehaceres_rev <- sdem_filtered %>%
  filter(sex == 2) %>%
  summarise(
    n_num    = sum(quehaceres & !is.na(hrs_quehaceres)),
    n_den    = n(),
    n_codigo = sum(quehaceres & is.na(hrs_quehaceres))
  )

# CV del promedio anual (error estándar combinado entre el promedio)
Hrs_quehaceres_cv <- Hrs_quehaceres %>%
  mutate(hrs_quehaceres_cv = hrs_quehaceres_se / hrs_quehaceres)

print(cbind(Hrs_quehaceres_cv, Hrs_quehaceres_rev))
if (Hrs_quehaceres_cv$hrs_quehaceres_cv * 100 > 30) {
  cat("OJO: CV mayor a 30%, no publicable\n")
} else if (Hrs_quehaceres_cv$hrs_quehaceres_cv * 100 > 15) {
  cat("OJO: CV entre 15% y 30%, precisión moderada\n")
}

# Lo mismo por alcaldía
Hrs_quehaceres_n_mun <- sdem_filtered %>%
  filter(sex == 2, quehaceres) %>%
  group_by(mun) %>%
  summarise(n = sum(!is.na(hrs_quehaceres)))

Hrs_quehaceres_cv_mun <- Hrs_quehaceres_x_alcaldia %>%
  left_join(Hrs_quehaceres_n_mun, by = "mun") %>%
  mutate(hrs_quehaceres_cv = hrs_quehaceres_se / hrs_quehaceres * 100,
         precision = case_when(hrs_quehaceres_cv > 30 ~ "No publicable",
                               hrs_quehaceres_cv > 15 ~ "Moderada",
                               TRUE               ~ "Buena"))

print(Hrs_quehaceres_cv_mun, n = 16)

# GUARDAR RESULTADOS

write.csv(
  Hrs_quehaceres,
  file = file.path(out_dir, "hrs_quehaceres_CDMX_2025.csv"),
  row.names = FALSE
)

write.csv(
  Hrs_quehaceres_x_alcaldia,
  file = file.path(out_dir, "hrs_quehaceres_x_alcaldia_2025.csv"),
  row.names = FALSE
)

write.csv(
  Hrs_quehaceres_nacional,

  row.names = FALSE
)
